rules_version = '2';

// Angkor Formal – Firestore security rules.
// Roles live in /users/{uid}. Everything is denied unless a rule below allows it.
service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- helpers ----------
    function signedIn() { return request.auth != null; }
    function me() { return get(/databases/$(database)/documents/users/$(request.auth.uid)).data; }
    function meExists() { return exists(/databases/$(database)/documents/users/$(request.auth.uid)); }
    function active() { return signedIn() && meExists() && me().active == true; }
    function isOwner() { return active() && me().role == 'owner'; }
    function isManager() { return active() && me().role == 'manager'; }
    function isCashier() { return active() && me().role == 'cashier'; }
    function isPartner() { return active() && me().role == 'shareholder'; }
    // The super admin is the very first person set up. /setup/state can never be changed, so this cannot be moved to anyone else from inside the app.
    function superUid() { return get(/databases/$(database)/documents/setup/state).data.firstOwner; }
    function isSuper() { return active() && request.auth.uid == superUid(); }
    function isLead() { return isOwner() || isManager(); }
    function inStore(sid) { return isOwner() || (active() && (sid in me().storeIds)); }
    function leadOf(sid) { return isOwner() || (isManager() && (sid in me().storeIds)); }
    // TESTING ONLY: lets the owner wipe test data while settings/testMode.enabled is true. Remove with the Reset button.
    function testReset() { return isSuper() && exists(/databases/$(database)/documents/settings/testMode) && get(/databases/$(database)/documents/settings/testMode).data.enabled == true; }
    function onlyChanges(keys) { return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys); }

    // ---------- people ----------
    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isOwner());
      // Only owners create or change people. Nobody deletes (deactivate instead).
      // Only the super admin creates owners or changes an owner. Other owners may manage managers, cashiers and shareholders, never the super admin.
      allow create: if isSuper() || (isOwner() && request.resource.data.role != 'owner');
      allow update: if isSuper()
        || (isOwner() && uid != superUid() && resource.data.role != 'owner' && request.resource.data.role != 'owner')
        || (isOwner() && request.auth.uid == uid && request.resource.data.role == resource.data.role && request.resource.data.active == resource.data.active);
      // First owner only: allowed when /setup/state is created in the same batch by this very person.
      allow create: if signedIn() && request.auth.uid == uid
        && request.resource.data.role == 'owner' && request.resource.data.active == true
        && !exists(/databases/$(database)/documents/setup/state)
        && existsAfter(/databases/$(database)/documents/setup/state)
        && getAfter(/databases/$(database)/documents/setup/state).data.firstOwner == uid;
      // An owner may remove someone (never themselves). The app only allows it when they have no history.
      allow delete: if (isSuper() && request.auth.uid != uid) || (isOwner() && uid != superUid() && resource.data.role != 'owner');
    }

    // One-time setup: the very first person to sign in creates the first owner.
    match /setup/{id} {
      allow get: if true;
      allow create: if signedIn() && id == 'state' && request.resource.data.firstOwner == request.auth.uid;
      allow update, delete: if false;
    }
    match /meta/{id} { allow read: if active(); allow write: if isOwner(); }

    // Till login screen needs names only, never roles or PINs.
    match /staffDirectory/{id} {
      allow read: if true;
      allow write: if isOwner();
    }

    // ---------- stores and settings ----------
    // Names only, so the sign-in screen can list stores before anyone is signed in.
    match /publicStores/{id} { allow read: if true; allow write: if isOwner(); }
    match /stores/{sid} {
      allow read: if active();
      allow write: if isOwner();
    }
    match /brands/{id} { allow read: if active(); allow write: if isOwner(); }
    match /types/{id}   { allow read: if active(); allow write: if isOwner(); }
    match /colours/{id} { allow read: if active(); allow write: if isOwner(); }
    match /settings/{id} {
      allow read: if active();
      allow write: if (id == 'rate' && isLead()) || isOwner();
    }

    // ---------- catalogue (no cost in here) ----------
    match /designs/{id} { allow read: if active(); allow write: if isOwner(); }
    match /items/{id}   { allow read: if active(); allow write: if isOwner(); }
    // Cost lives apart so cashiers can never read it.
    match /costs/{id}   { allow read: if isLead() || isPartner(); allow write: if isOwner(); }

    // ---------- stock ----------
    // Document id: <storeId>_<itemId>
    match /stock/{key} {
      allow read: if active();
      // Managers and owners adjust stock of their own store.
      allow create, update: if leadOf(request.resource.data.storeId);
      // A cashier may only lower stock, never below zero, and only together with a sale
      // written in the same batch.
      allow update: if isCashier()
        && inStore(resource.data.storeId)
        && request.resource.data.storeId == resource.data.storeId
        && request.resource.data.onHand >= 0
        && request.resource.data.onHand < resource.data.onHand
        && request.resource.data.lastSaleId is string
        && existsAfter(/databases/$(database)/documents/sales/$(request.resource.data.lastSaleId));
      allow delete: if testReset();
    }
    // Every change to stock is also written here. Records can only be added.
    match /movements/{id} {
      allow read: if isLead();
      allow create: if active()
        && inStore(request.resource.data.storeId)
        && request.resource.data.userId == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }

    // ---------- selling ----------
    match /counters/{sid} {
      allow delete: if testReset();
      allow read: if active() && inStore(int(sid));
      allow create: if active() && inStore(int(sid)) && request.resource.data.seq == 1;
      allow update: if active() && inStore(int(sid)) && request.resource.data.seq == resource.data.seq + 1;
    }

    match /sales/{id} {
      allow read: if active() && (isLead() ? inStore(resource.data.storeId) : (inStore(resource.data.storeId) && resource.data.cashierId == request.auth.uid));
      allow create: if active()
        && inStore(request.resource.data.storeId)
        && request.resource.data.cashierId == request.auth.uid
        && request.resource.data.status == 'completed'
        // A discount above the store's limit needs a manager's approval that already exists
        // (managers and owners may discount freely in their own store).
        && (leadOf(request.resource.data.storeId)
            || (request.resource.data.discountCents * 100 <= request.resource.data.subtotalCents
                  * get(/databases/$(database)/documents/stores/$(string(request.resource.data.storeId))).data.switches.cashier_discount_limit_percent
                && (!get(/databases/$(database)/documents/stores/$(string(request.resource.data.storeId))).data.switches.pin_every_discount
                    || request.resource.data.discountCents == 0))
            || (request.resource.data.approvalId is string
                && get(/databases/$(database)/documents/approvals/$(request.resource.data.approvalId)).data.status == 'approved'
                && get(/databases/$(database)/documents/approvals/$(request.resource.data.approvalId)).data.storeId == request.resource.data.storeId));
      // Voiding needs a manager or owner, and only the void fields may change.
      allow update: if leadOf(resource.data.storeId)
        && resource.data.status == 'completed'
        && request.resource.data.status == 'voided'
        && onlyChanges(['status', 'voidedBy', 'voidReason', 'voidedTs']);
      allow delete: if testReset();
    }

    // Returns are always written by a manager or owner (a manager signs in at the till to approve).
    match /returns/{id} {
      allow read: if active() && (isLead() ? inStore(resource.data.storeId) : resource.data.requestedBy == request.auth.uid);
      allow create: if leadOf(request.resource.data.storeId);
      allow update: if false;
      allow delete: if testReset();
    }

    match /shifts/{id} {
      allow read: if active() && (isLead() ? inStore(resource.data.storeId) : resource.data.userId == request.auth.uid);
      allow create: if active() && inStore(request.resource.data.storeId) && request.resource.data.userId == request.auth.uid;
      // Close your own shift, or a lead reviews someone else's.
      allow update: if active() && inStore(resource.data.storeId)
        && ((resource.data.userId == request.auth.uid && resource.data.closedTs == null)
            || (isLead() && resource.data.userId != request.auth.uid && onlyChanges(['reviewedBy', 'reviewedTs'])));
      allow delete: if testReset();
    }

    // Daily totals the dashboard reads. Cashiers may add to them but never read them.
    match /dailyStats/{key} {
      allow read: if active() && isLead() && inStore(resource.data.storeId);
      allow create, update: if active() && inStore(request.resource.data.storeId);
      allow delete: if testReset();
    }

    // Pieces sold per product per day, no money. Every active person in the branch can read it so the
    // best-seller ranking works for cashiers too.
    match /itemStats/{key} {
      allow read: if active() && (inStore(resource.data.storeId) || isPartner());
      allow create, update: if active() && inStore(request.resource.data.storeId);
      allow delete: if testReset();
    }

    // Monthly totals (for the year view).
    match /monthStats/{key} {
      allow read: if active() && (isPartner() || (isLead() && inStore(resource.data.storeId)));
      allow create, update: if active() && inStore(request.resource.data.storeId);
      allow delete: if testReset();
    }

    // Manager approvals (replace PIN entry on the cashier's screen).
    match /approvals/{id} {
      allow read: if active() && (resource.data.requestedBy == request.auth.uid || leadOf(resource.data.storeId));
      allow create: if active() && inStore(request.resource.data.storeId)
        && request.resource.data.requestedBy == request.auth.uid
        && request.resource.data.status == 'pending';
      // Nobody approves their own request.
      allow update: if leadOf(resource.data.storeId)
        && resource.data.requestedBy != request.auth.uid
        && resource.data.status == 'pending'
        && request.resource.data.status in ['approved', 'rejected']
        && onlyChanges(['status', 'decidedBy', 'decidedTs']);
      allow delete: if testReset();
    }

    match /customers/{id} {
      allow read, create: if active();
      allow update: if active();
      allow delete: if isOwner();
    }

    // ---------- transfers between stores ----------
    match /transfers/{id} {
      allow read: if active() && (isOwner() || inStore(resource.data.fromStoreId) || inStore(resource.data.toStoreId));
      allow create: if active()
        && (inStore(request.resource.data.fromStoreId) || inStore(request.resource.data.toStoreId))
        && request.resource.data.requestedBy == request.auth.uid
        && request.resource.data.status == 'requested';
      allow update: if active() && (
        // approve or reject: lead of the sending store, never the requester
        (resource.data.status == 'requested'
          && leadOf(resource.data.fromStoreId)
          && resource.data.requestedBy != request.auth.uid
          && request.resource.data.status in ['approved', 'awaiting_owner', 'rejected'])
        // owner approves large transfers
        || (resource.data.status == 'awaiting_owner' && isOwner() && resource.data.requestedBy != request.auth.uid
            && request.resource.data.status in ['approved', 'rejected'])
        // sending store marks as sent
        || (resource.data.status == 'approved' && leadOf(resource.data.fromStoreId)
            && request.resource.data.status == 'in_transit')
        // receiving store confirms
        || (resource.data.status == 'in_transit' && leadOf(resource.data.toStoreId)
            && request.resource.data.status in ['received', 'discrepancy'])
        // the owner settles a difference
        || (resource.data.status == 'discrepancy' && isOwner() && request.resource.data.status == 'resolved')
        // cancel before it is sent (stock already set aside can only be released by a lead of the sending store)
        || (resource.data.status == 'requested'
            && (resource.data.requestedBy == request.auth.uid || leadOf(resource.data.fromStoreId))
            && request.resource.data.status == 'cancelled')
        || (resource.data.status in ['awaiting_owner', 'approved'] && leadOf(resource.data.fromStoreId)
            && request.resource.data.status == 'cancelled')
      );
      allow delete: if testReset();
    }

    // ---------- expenses ----------
    match /expenseCategories/{id} { allow read: if active(); allow write: if isOwner(); }
    match /expenses/{id} {
      allow read: if active() && (isPartner() || (isLead() && (resource == null || inStore(resource.data.storeId))));
      allow create: if leadOf(request.resource.data.storeId) && request.resource.data.createdBy == request.auth.uid;
      allow delete: if leadOf(resource.data.storeId);
      allow update: if false;
    }

    // Receipt photos live apart from the expense list so the list stays light.
    match /expensePhotos/{id} {
      allow read: if active() && isLead() && (resource == null || inStore(resource.data.storeId));
      allow create: if leadOf(request.resource.data.storeId);
      allow delete: if leadOf(resource.data.storeId);
      allow update: if false;
    }

    // ---------- shareholder capital: owner writes, owner and shareholders read; payments are never edited ----------
    match /shareholders/{id} {
      allow read: if isOwner() || isPartner();
      allow write: if isOwner();
    }
    match /capitalPayments/{id} {
      allow read: if isOwner() || isPartner();
      allow create: if isOwner() && request.resource.data.createdBy == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }

    // ---------- deposits, alteration jobs, stock counts ----------
    match /balances/{id} {
      allow read: if active() && inStore(resource.data.storeId);
      allow create: if active() && inStore(request.resource.data.storeId) && request.resource.data.createdBy == request.auth.uid
        && existsAfter(/databases/$(database)/documents/sales/$(request.resource.data.saleId));
      allow update: if active() && inStore(resource.data.storeId) && !resource.data.settled
        && onlyChanges(['paidCents', 'balanceCents', 'settled', 'settledTs', 'lastTs'])
        && request.resource.data.balanceCents >= 0 && request.resource.data.balanceCents < resource.data.balanceCents;
      allow delete: if testReset();
    }
    match /balancePayments/{id} {
      allow read: if active() && inStore(resource.data.storeId) && (isLead() || resource.data.userId == request.auth.uid);
      allow create: if active() && inStore(request.resource.data.storeId) && request.resource.data.userId == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }
    match /alterations/{id} {
      allow read: if active() && inStore(resource.data.storeId);
      allow create: if active() && inStore(request.resource.data.storeId) && request.resource.data.createdBy == request.auth.uid;
      allow update: if active() && inStore(resource.data.storeId) && request.resource.data.storeId == resource.data.storeId;
      allow delete: if testReset();
    }
    match /stocktakes/{id} {
      allow read: if leadOf(resource.data.storeId);
      allow create: if leadOf(request.resource.data.storeId) && request.resource.data.createdBy == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }

    // ---------- attendance ----------
    // Clocking in or out must happen "now" (within 5 minutes of the server clock), so a time cannot be made up.
    // Managers can add or correct records for others in their own store, but not for themselves.
    match /attendance/{id} {
      allow read: if active() && (resource == null || resource.data.userId == request.auth.uid || (isLead() && inStore(resource.data.storeId)));
      allow create: if active() && (
        (request.resource.data.userId == request.auth.uid && inStore(request.resource.data.storeId)
          && request.resource.data.manual == false && request.resource.data.excused == false
          && request.resource.data.inTs >= request.time.toMillis() - 300000 && request.resource.data.inTs <= request.time.toMillis() + 300000)
        || (leadOf(request.resource.data.storeId) && (isOwner() || request.resource.data.userId != request.auth.uid)));
      allow update: if active() && (
        (resource.data.userId == request.auth.uid && onlyChanges(['outTs']) && resource.data.outTs == null
          && request.resource.data.outTs >= request.time.toMillis() - 300000 && request.resource.data.outTs <= request.time.toMillis() + 300000)
        || (leadOf(resource.data.storeId) && (isOwner() || resource.data.userId != request.auth.uid)));
      allow delete: if leadOf(resource.data.storeId) && (isOwner() || resource.data.userId != request.auth.uid);
    }

    // ---------- profit share: a month is closed once (only the super admin can reopen it); payouts are never edited ----------
    match /profitCloses/{id} {
      allow read: if isOwner() || isPartner();
      allow create: if isOwner() && request.resource.data.closedBy == request.auth.uid;
      allow update: if false;
      allow delete: if isSuper() || testReset();
    }
    match /profitPayouts/{id} {
      allow read: if isOwner() || isPartner();
      allow create: if isOwner() && request.resource.data.createdBy == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }

    // ---------- activity log ----------
    match /auditLog/{id} {
      allow read: if isLead();
      allow create: if active() && request.resource.data.userId == request.auth.uid;
      allow update: if false;
      allow delete: if testReset();
    }

    match /{document=**} { allow read, write: if false; }
  }
}
